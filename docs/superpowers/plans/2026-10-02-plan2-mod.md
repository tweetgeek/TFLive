## Global Constraints

- Mod: `modId = "tflive"`, katalog `mod/tflive/`, skrypty w `mod/tflive/content/tflive/`. Moduły wczytywane wyłącznie przez `ug_require "tflive::/tflive/<ścieżka>.tl"` (krok 0 #20).
- Mod nie ma `io`; jedynym wyjściem jest `debugPrint(linia)`. Nigdy nie wypisuje tabel (`debugPrint(tabela)`) — tylko gotowe linie tekstu.
- Format linii: `[TFLIVE] 1 <saveId> <sessionId> <seq> <gameTime> <sekcja> <część>/<części> <json>`; maks. ~90 000 bajtów JSON na linię (`wire.MAX_LINE_BYTES`); sekcje tablicowe dzielone na części.
- Każda nowa sesja modu zaczyna się od `meta`; brakujące wartości to jawne `null` (sentinel `json.NULL`); liczby w nagłówku całkowite.
- Krotki dokładnie wg kontraktu: `lines` 11 pól (ostatnie `opsM`), `stops` = `[id,name,[[cargo,count,bad]],veryBad01]` (Task 1 tego planu).
- Stan modu wyłącznie w `GameScriptState` (pula wątków symulacji, osobne stany Lua). Trwałe: `saveId`; sesyjne: reszta, zerowana przy nowej sesji (przerwa `lastEmitWall` > 30 s).
- Interwały (czas rzeczywisty `os.time()`): `meta`, `finance` 10 s; `lines`, `vehicles`, `stops` 60 s; `industries`, `towns` 120 s. W pauzie (`dt == 0` lub `GAME_SPEED.speedup == 0`) tylko `meta` (co 10 s, `paused=1`). Sekcja jednoczęściowa z niezmienionym odciskiem nie jest wysyłana (poza `meta`).
- Enumy gry w runtime są liczbami — nazwy wyłącznie przez porównanie z `api.type["enum"].*` (krok 0 #29).
- Granice miesięcy: arytmetycznie z `getCalendarDate` i stałej długości dnia (`GAME_SPEED.millisPerDay`, krok 0 #21) — **nie** przez iterowanie `GameTime.dates` (koszt rośnie z wiekiem gry).
- Node ≥ 22.13, ESM, TypeScript strict (jak w planie 1). Commity: temat, JEDNA pusta linia, potem dokładnie:
  ```

  Claude-Session: https://claude.ai/code/session_01T91NpZcqyTDnQB7HUqAw4K
  ```
- Narzędzia deweloperskie (już zainstalowane na macOS użytkownika): `luajit`, `tl` 0.24.8 w `~/.luarocks` (Lua 5.5 z Homebrew nie uruchamia `tl` — zawsze przez LuaJIT, wzorzec `spike/tlcheck.sh`). Typy gry: `<TF3>/api/tealdef`, `<TF3>/base/tealdef` (`TF3_DIR`, domyślnie ścieżka Steam na macOS).

## Review Focus

- **Pole `nil` w środku krotki** (np. linia bez koloru, brak `firstLedger`): musi trafić do JSON jako `null`, nie skracać tablicy (dekoder odrzuca złą liczbę pól). Test w Task 2 (json) i Task 4 (golden z `NULL`-ami).
- **Restart gry / wczytanie zapisu**: `GameScriptState` wraca z zapisu z odciskami sekcji i starym `lastEmitWall` — mod musi rozpoznać nową sesję i wysłać wszystko od nowa, zaczynając od `meta`. Test w Task 3 (schedule) + weryfikacja w Task 9.
- **Pauza**: tylko `meta` z `paused=1`, co 10 s; pozostałe sekcje i backfill wstrzymane. Weryfikacja w Task 9.
- **Duża sekcja** (setki pojazdów): podział na części < 90 KB, wszystkie części z tym samym `seq`. Test w Task 3 (chunkRows) i Task 4 (golden z wieloma częściami).
- **Błąd jednego kolektora** (np. nieznane pole API): sekcja pominięta, wpis w `meta.errors`, pozostałe sekcje działają. Kod w Task 8, weryfikacja w Task 9.

```ts
export const overloadedStop: Rule = ({ snap, config }) =>
  snap.stops.flatMap((s) => {
    const total = s.waiting.reduce((sum, w) => sum + w.count, 0);
    const bad = s.waiting.reduce((sum, w) => sum + w.bad, 0);
    const ratio = total > 0 ? bad / total : 0;
    if (!s.veryBad && ratio < config.overloadedBadRatio) return [];
    return [{
      type: 'overloaded-stop' as const,
      category: 'lines' as const,
      severity: 'warn' as const,
      entityType: 'stop' as const,
      entityId: s.id,
      title: `Przeciążony przystanek: ${s.name}`,
      detail: s.veryBad ? 'Gra ocenia obsługę oczekujących jako bardzo złą' : `Źle obsłużeni: ${Math.round(ratio * 100)}% oczekujących`,
    }];
  });
```

```lua
-- Runner testów czystych modułów modu (LuaJIT). Moduły: mod/build/<nazwa>.lua (z `tl gen`).
local here = debug.getinfo(1, "S").source:sub(2):match("(.*/)") or "./"
local M = {}
M.root = here .. "../"
function M.load(name)
	return dofile(M.root .. "build/" .. name .. ".lua")
end

local function deepEq(a, b)
	if type(a) ~= type(b) then return false end
	if type(a) ~= "table" then return a == b end
	for k, v in pairs(a) do if not deepEq(v, b[k]) then return false end end
	for k, _ in pairs(b) do if a[k] == nil then return false end end
	return true
end
local function show(v)
	if type(v) ~= "table" then return tostring(v) end
	local parts = {}
	for k, x in pairs(v) do parts[#parts + 1] = tostring(k) .. "=" .. show(x) end
	return "{" .. table.concat(parts, ",") .. "}"
end

local passed, failed = 0, 0
function test(name, fn)
	local ok, err = pcall(fn)
	if ok then passed = passed + 1 else failed = failed + 1; print("FAIL " .. name .. ": " .. tostring(err)) end
end
function eq(actual, expected, msg)
	if not deepEq(actual, expected) then error((msg or "eq") .. ": expected " .. show(expected) .. ", got " .. show(actual), 2) end
end
function ok(cond, msg)
	if not cond then error(msg or "ok", 2) end
end

_G.M = M
for _, file in ipairs({ "json_test", "wire_test", "months_test", "schedule_test", "golden" }) do
	local path = here .. file .. ".lua"
	local f = io.open(path, "r")
	if f then f:close(); dofile(path) end
end
print(string.format("lua tests: %d passed, %d failed", passed, failed))
os.exit(failed == 0 and 0 or 1)
```

`mod/tflive/content/tflive/lib/json.tl`:
```lua
-- Koder JSON dla linii [TFLIVE]. Czysty moduł (bez API gry).
local record Json
	NULL : table
	encode : function(any) : string
end

local json : Json = { NULL = {} }

local function escape(s : string) : string
	local out = s:gsub('[%c"\\]', function(c : string) : string
		if c == '"' then return '\\"' end
		if c == '\\' then return '\\\\' end
		if c == '\n' then return '\\n' end
		if c == '\r' then return '\\r' end
		if c == '\t' then return '\\t' end
		return string.format('\\u%04x', c:byte())
	end)
	return out
end

local function number(n : number) : string
	if n ~= n or n == math.huge or n == -math.huge then return 'null' end
	if n == math.floor(n) and math.abs(n) < 1e15 then return string.format('%.0f', n) end
	return string.format('%.6g', n)
end

local function isArray(t : table) : boolean
	for k, _ in pairs(t) do
		if type(k) ~= 'number' then return false end
	end
	return true
end

local encode : function(any) : string
encode = function(v : any) : string
	if v == nil or v == json.NULL then return 'null' end
	local tv = type(v)
	if tv == 'boolean' then return (v as boolean) and 'true' or 'false' end
	if tv == 'number' then return number(v as number) end
	if tv == 'string' then return '"' .. escape(v as string) .. '"' end
	if tv == 'table' then
		local t = v as table
		if isArray(t) then
			local arr = t as {any}
			local parts : {string} = {}
			for i = 1, #arr do parts[i] = encode(arr[i]) end
			return '[' .. table.concat(parts, ',') .. ']'
		end
		local obj = t as {string : any}
		local keys : {string} = {}
		for k, _ in pairs(obj) do table.insert(keys, k) end
		table.sort(keys)
		local parts : {string} = {}
		for _, k in ipairs(keys) do table.insert(parts, '"' .. escape(k) .. '":' .. encode(obj[k])) end
		return '{' .. table.concat(parts, ',') .. '}'
	end
	error('json: nieobsługiwany typ ' .. tv)
end
json.encode = encode

return json
```

```lua
-- Format linii [TFLIVE], podział sekcji na części, odcisk treści. Czysty moduł.
local record Wire
	SCHEMA_VERSION : integer
	MAX_LINE_BYTES : integer
	formatLine : function(string, string, integer, number, string, integer, integer, string) : string
	chunkRows : function({string}, integer) : {string}
	fingerprint : function(string) : string
end

local wire : Wire = { SCHEMA_VERSION = 1, MAX_LINE_BYTES = 90000 }

wire.formatLine = function(saveId : string, sessionId : string, seq : integer, gameTime : number, section : string, part : integer, parts : integer, json : string) : string
	return string.format('[TFLIVE] %d %s %s %d %.0f %s %d/%d %s', wire.SCHEMA_VERSION, saveId, sessionId, seq, gameTime, section, part, parts, json)
end

wire.chunkRows = function(rowsJson : {string}, maxBytes : integer) : {string}
	local out : {string} = {}
	local cur : {string} = {}
	local size = 2
	for _, r in ipairs(rowsJson) do
		if #cur > 0 and size + #r + 1 > maxBytes then
			table.insert(out, '[' .. table.concat(cur, ',') .. ']')
			cur = {}
			size = 2
		end
		table.insert(cur, r)
		size = size + #r + 1
	end
	table.insert(out, '[' .. table.concat(cur, ',') .. ']')
	return out
end

wire.fingerprint = function(s : string) : string
	local h = 5381
	local n = #s
	local i = 1
	while i <= n do
		local j = math.min(i + 255, n)
		local bytes = { s:byte(i, j) }
		for k = 1, #bytes do h = (h * 33 + bytes[k]) % 4294967296 end
		i = j + 1
	end
	return string.format('%.0f:%d', h, n)
end

return wire
```

```lua
end

	return y * 12 + (m - 1)
end

months.lastMonths = function(now : integer, dayMs : integer, n : integer, cal : Calendar) : {{integer}}
	local out : {{integer}} = {}
	local y, m, d = cal(now)
	local from = math.floor(now / dayMs) * dayMs - (d - 1) * dayMs
	local to = now
	while #out < n do
		if from < 0 then from = 0 end
		table.insert(out, 1, { months.index(y, m), from, to })
		if from <= 0 then break end
		to = from
		local py, pm, pd = cal(from - 1)
		y, m = py, pm
		from = math.floor((from - 1) / dayMs) * dayMs - (pd - 1) * dayMs
	end
	return out
end
```

```lua
-- TFLive: wypisuje stan Rozgrywki do logu gry jako linie [TFLIVE] (spec §3, §7; ADR 0003).
local record Json
	NULL : table
	encode : function(any) : string
end
local record Wire
	MAX_LINE_BYTES : integer
	formatLine : function(string, string, integer, number, string, integer, integer, string) : string
	chunkRows : function({string}, integer) : {string}
	fingerprint : function(string) : string
end
local record Schedule
	ORDER : {string}
	INTERVALS : {string : integer}
	isNewSession : function(integer, integer) : boolean
	isDue : function(integer, integer, integer) : boolean
	newId : function(number) : string
end
local record Collect
	gameTime : function() : integer
	isPaused : function(number) : boolean
	meta : function(number, integer, {string}) : {string : any}
	finance : function() : {string : any}
	backfill : function(integer) : {string : any}
	lines : function({string : integer}) : {any}
	vehicles : function() : {any}
	stops : function() : {any}
	industries : function() : {any}
	towns : function() : {any}
end

local json = ug_require "tflive::/tflive/lib/json.tl" as Json
local wire = ug_require "tflive::/tflive/lib/wire.tl" as Wire
local schedule = ug_require "tflive::/tflive/lib/schedule.tl" as Schedule
local collect = ug_require "tflive::/tflive/collect.tl" as Collect

local MAX_ERRORS <const> = 5

local record SectionState
	lastWall : integer
	fp : string
end

local record ModState
	saveId : string            -- trwałe (zapisywane z grą)
	sessionId : string         -- reszta: sesyjne, zerowane przy nowej sesji
	lastEmitWall : integer
	seq : integer
	bytes : integer
	sections : {string : SectionState}
	backfillIndex : integer
	departures : {string : integer}
	errors : {string}
end

local function addError(s : ModState, msg : string)
	table.insert(s.errors, msg)
	while #s.errors > MAX_ERRORS do table.remove(s.errors, 1) end
end

local function emit(s : ModState, section : string, gameTime : integer, parts : {string})
	s.seq = s.seq + 1
	for i, p in ipairs(parts) do
		local line = wire.formatLine(s.saveId, s.sessionId, s.seq, gameTime, section, i, #parts, p)
		debugPrint(line)
		s.bytes = s.bytes + #line + 1
	end
end

local function collectSection(s : ModState, name : string, dt : number) : any
	if name == 'meta' then return collect.meta(dt, s.bytes, s.errors) end
	if name == 'finance' then return collect.finance() end
	if name == 'lines' then return collect.lines(s.departures) end
	if name == 'vehicles' then return collect.vehicles() end
	if name == 'stops' then return collect.stops() end
	if name == 'industries' then return collect.industries() end
	if name == 'towns' then return collect.towns() end
	error('nieznana sekcja ' .. name)
end

local function emitSection(s : ModState, name : string, dt : number, gameTime : integer, wall : integer)
	local errMsg : string = nil
	local data : any = nil
	local ok = xpcall(function() data = collectSection(s, name, dt) end, function(e : any) : any
		errMsg = tostring(e)
		return e
	end)
	local sec = s.sections[name] or {}
	sec.lastWall = wall
	s.sections[name] = sec
	if not ok then
		addError(s, name .. ': ' .. tostring(errMsg))
		return
	end
	local parts : {string}
	if name == 'meta' or name == 'finance' then
		parts = { json.encode(data) }
	else
		local rowsJson : {string} = {}
		for i, r in ipairs(data as {any}) do rowsJson[i] = json.encode(r) end
		parts = wire.chunkRows(rowsJson, wire.MAX_LINE_BYTES)
	end
	if name ~= 'meta' and #parts == 1 then
		local fp = wire.fingerprint(parts[1])
		if sec.fp == fp then return end
		sec.fp = fp
	end
	emit(s, name, gameTime, parts)
	if name == 'meta' then s.errors = {} end
end

local function emitBackfill(s : ModState, gameTime : integer)
	local errMsg : string = nil
	local data : {string : any} = nil
	local ok = xpcall(function() data = collect.backfill(s.backfillIndex) end, function(e : any) : any
		errMsg = tostring(e)
		return e
	end)
	if not ok then
		addError(s, 'backfill: ' .. tostring(errMsg))
		s.backfillIndex = s.backfillIndex + 1
		return
	end
	if data == nil then
		s.backfillIndex = 0
		return
	end
	emit(s, 'backfill', gameTime, { json.encode(data) })
	s.backfillIndex = s.backfillIndex + 1
end

local function startSession(s : ModState, wall : integer)
	s.sessionId = schedule.newId(os.clock() * 1000 + wall)
	s.seq = 0
	s.bytes = 0
	s.sections = {}
	s.backfillIndex = 1
	s.departures = {}
	s.errors = {}
end

local ret : GameScript<ModState> = {
	update = function(_userParams : table, state : GameScriptState<ModState>, dt : number) : any
		local s = state:get() or {}
		local wall = os.time()
		if not s.saveId then s.saveId = schedule.newId(os.clock() + wall) end
		if schedule.isNewSession(s.lastEmitWall, wall) or not s.sessionId then startSession(s, wall) end
		local paused = collect.isPaused(dt)
		local gameTime = collect.gameTime()
		local emitted = false
		for _, name in ipairs(schedule.ORDER) do
			if name == 'meta' or not paused then
				local sec = s.sections[name]
				if schedule.isDue(sec and sec.lastWall, wall, schedule.INTERVALS[name]) then
					emitSection(s, name, dt, gameTime, wall)
					emitted = true
				end
			end
		end
		if not paused and s.backfillIndex and s.backfillIndex > 0 then
			emitBackfill(s, gameTime)
		end
		if emitted then s.lastEmitWall = wall end
		state:set(s)
	end,
}
```

```ts
import { cpSync, existsSync, mkdirSync, readdirSync, rmSync } from 'node:fs';
import { homedir } from 'node:os';
import { join } from 'node:path';
import { fileURLToPath } from 'node:url';
import { steamUserdataRoots, TF3_APP_ID } from '../src/paths';

export function modsDirs(roots: string[]): string[] {
  const out: string[] = [];
  for (const root of roots) {
    let ids: string[];
    try {
      ids = readdirSync(root);
    } catch {
      continue;
    }
    for (const id of ids) {
      const local = join(root, id, TF3_APP_ID, 'local');
      if (existsSync(local)) out.push(join(local, 'mods'));
    }
  }
  return out;
}

export function installMod(src: string, targets: string[], opts: { dryRun?: boolean } = {}): string[] {
  const done: string[] = [];
  for (const mods of targets) {
    const dest = join(mods, 'tflive');
    if (!opts.dryRun) {
      mkdirSync(mods, { recursive: true });
      rmSync(dest, { recursive: true, force: true });
      cpSync(src, dest, { recursive: true });
    }
    done.push(dest);
  }
  return done;
}
```
