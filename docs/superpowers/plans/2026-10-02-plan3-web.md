# Plan 3: interfejs web (Vue) — plan implementacji

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Dashboard w przeglądarce (tablet w poziomie, komputer, telefon), który na żywo pokazuje stan Rozgrywki z serwera TFLive: Przegląd, Alerty (z Wyciszaniem), Linie, Pojazdy, Przemysł, Miasta — zgodnie z zaakceptowaną makietą.

**Architecture:** Workspace `web/` (Vue 3 + Vite + TypeScript). Jeden reaktywny magazyn (`createStore`) trzyma stan z WebSocketa `/ws` (pełny `HubState` przy każdej zmianie), łączy się ponownie z backoffem i obsługuje Wyciszenia przez REST. Czyste funkcje (`lib/format.ts`, `lib/emoji.ts`, `lib/derive.ts`) liczą wartości do wyświetlenia. Widoki to komponenty SFC przełączane `#hash` (bez routera). Zbudowany `web/dist` serwuje istniejący serwer Fastify.

**Tech Stack:** Vue 3.5, Vite 8, `@vitejs/plugin-vue` 6, TypeScript 5.9, `vue-tsc` 3, Vitest (już w repo) + `@vue/test-utils` 2 + `jsdom` (testy komponentów), czcionki Google Fonts (Barlow Semi Condensed, IBM Plex Mono).

**Spec:** `docs/superpowers/specs/2026-10-01-tflive-dashboard-design.md` (§6 UI, §5 reguły i Wyciszenia, §9 błędy). Słownik: `CONTEXT.md`. Makieta (zaakceptowana przez użytkownika, z emoji): https://claude.ai/artifact/FA9PQpBJZgurgsxPRQEvjS — wygląd opisany w Global Constraints poniżej.

- Cele dotykowe ≥ 44 px; prawdziwe `<button>`, `<a href>`, `<input>`+`<label>`; kontrast tekstu ≥ 4.5:1.

## Review Focus

- **Zerwane połączenie / restart serwera**: baner „Brak połączenia z serwerem”, ponowne łączenie z rosnącym odstępem (1→2→5→10→30 s), po powrocie świeży stan. Test w Task 3.
- **Pusty stan** (serwer bez danych z gry: `meta === null`, puste listy): każdy widok pokazuje komunikat „Czekam na dane z gry…”, bez wyjątków przy dzieleniu przez 0 / `null`. Testy w Task 5–9 (fixture `emptyState`).
- **Pola `null`** (np. `firstLedgerTime`, `lastServedTime`, `supplied`, `color`): wyświetlane jako „—” lub pominięte, nigdy „null”/„NaN”. Testy w Task 2 (format) i widokach.
- **Wyciszony alert**: domyślnie ukryty w Alertach i na Przeglądzie; widoczny po „Pokaż wyciszone” z przyciskiem „Przywróć”, który usuwa właściwe Wyciszenie (dopasowanie po typie alertu i encji). Test w Task 6.
- **Wąski ekran (telefon)**: dolny pasek nawigacji, brak poziomego przewijania strony (tylko tabele). Test w Task 4 (klasa układu wg szerokości) + kontrola wizualna w Task 10.

```ts
import type { GameDate } from '@tflive/shared';

export const MONTHS_SHORT = ['sty', 'lut', 'mar', 'kwi', 'maj', 'cze', 'lip', 'sie', 'wrz', 'paź', 'lis', 'gru'] as const;
const intFmt = new Intl.NumberFormat('pl-PL', { maximumFractionDigits: 0, useGrouping: 'always' } as Intl.NumberFormatOptions);

const ok = (n: number | null | undefined): n is number => typeof n === 'number' && Number.isFinite(n);

export function fmtInt(n: number | null | undefined): string {
  return ok(n) ? intFmt.format(Math.round(n)) : '—';
}

export function fmtSigned(n: number | null | undefined): string {
  if (!ok(n)) return '—';
  const r = Math.round(n);
  if (r === 0) return '0';
  return (r > 0 ? '+' : '−') + intFmt.format(Math.abs(r));
}

export function fmtPct(r: number | null | undefined): string {
  return ok(r) ? `${Math.round(r * 100)}%` : '—';
}

export function fmtDecimal(n: number | null | undefined, digits = 1): string {
  return ok(n) ? n.toFixed(digits).replace('.', ',') : '—';
}

export function fmtGameDate(d: GameDate | null | undefined): string {
  return d ? `${d.d} ${MONTHS_SHORT[d.m - 1]} ${d.y}` : '—';
}

export function fmtMonthIndex(i: number): string {
  return `${MONTHS_SHORT[i % 12]} ${Math.floor(i / 12)}`;
}

export function fmtMBh(n: number | null | undefined): string {
  return `${fmtDecimal(n)} MB/h`;
}
```

```ts
import type { Alert, Finance, GameDate, Line, Meta, Town, Vehicle } from '@tflive/shared';

export const RAMP_UP_DAYS = 180;
export const SERVED_DAYS = 365;

export function reserveMonths(f: Finance | null): number | null {
  if (!f || f.runningCosts12m <= 0) return null;
  return f.balance / (f.runningCosts12m / 12);
}

export function populationTotal(towns: Town[]): number {
  return towns.reduce((s, t) => s + t.population, 0);
}

export function bestVehicle(vs: Vehicle[]): Vehicle | null {
  let best: Vehicle | null = null;
  for (const v of vs) {
    if (v.operatingResult12m === null) continue;
    if (!best || v.operatingResult12m > (best.operatingResult12m ?? -Infinity)) best = v;
  }
  return best;
}

export function vehicleAgeYears(v: Vehicle, gameTime: number, dayMs: number): number | null {
  if (!v.parts.length || dayMs <= 0) return null;
  const oldest = Math.min(...v.parts.map((p) => p.purchaseTime));
  return (gameTime - oldest) / (365 * dayMs);
}

export function vehicleCondition(v: Vehicle): number | null {
  return v.parts.length ? Math.min(...v.parts.map((p) => p.maintenanceState)) : null;
}

export function lifespanUsage(v: Vehicle, gameTime: number): number | null {
  const xs = v.parts.filter((p) => p.lifespanSec > 0).map((p) => (gameTime - p.purchaseTime) / (p.lifespanSec * 1000));
  return xs.length ? Math.max(...xs) : null;
}

export function lineLoadNow(line: Line, vehicles: Vehicle[]): number | null {
  let load = 0;
  let cap = 0;
  for (const v of vehicles) {
    if (v.lineId !== line.id) continue;
    load += v.load;
    cap += v.capacity;
  }
  return cap > 0 ? load / cap : null;
}

export function isRampUp(line: Line, gameTime: number, dayMs: number): boolean {
  return line.firstLedgerTime !== null && gameTime - line.firstLedgerTime < RAMP_UP_DAYS * dayMs;
}

export function isServed(lastServedTime: number | null, gameTime: number, dayMs: number): boolean {
  return lastServedTime !== null && gameTime - lastServedTime <= SERVED_DAYS * dayMs;
}

export function gameDateAt(t: number, meta: Meta): GameDate {
  const daysBack = Math.floor(meta.gameTime / meta.dayMs) - Math.floor(t / meta.dayMs);
  const d = new Date(Date.UTC(meta.date.y, meta.date.m - 1, meta.date.d - daysBack));
  return { y: d.getUTCFullYear(), m: d.getUTCMonth() + 1, d: d.getUTCDate() };
}

export function alertsFor(alerts: Alert[], entityType: Alert['entityType'], entityId: number): Alert[] {
  return alerts.filter((a) => a.entityType === entityType && a.entityId === entityId);
}
```

```ts
import { emptySnapshot, type Alert, type Mute, type MuteKind } from '@tflive/shared';
import { inject, reactive, type InjectionKey } from 'vue';
import type { HubState, HubStatus, WireMessage } from './types';

export type Conn = 'connecting' | 'open' | 'closed';

export interface WebSocketLike {
  onopen: (() => void) | null;
  onmessage: ((ev: { data: unknown }) => void) | null;
  onclose: (() => void) | null;
  onerror: (() => void) | null;
  close(): void;
}

export interface StoreDeps {
  url: string;
  wsFactory: (url: string) => WebSocketLike;
  fetch: (input: string, init?: RequestInit) => Promise<Response>;
  setTimeout?: (fn: () => void, ms: number) => unknown;
}

export interface AppState extends HubState {
  conn: Conn;
  mutes: Mute[];
  received: boolean;
}

export interface Store {
  state: AppState;
  connect(): void;
  close(): void;
  refreshMutes(): Promise<void>;
  mute(alert: Alert, kind: MuteKind): Promise<void>;
  unmute(alert: Alert): Promise<void>;
}

export const BACKOFF_MS = [1000, 2000, 5000, 10000, 30000] as const;
export const STORE_KEY: InjectionKey<Store> = Symbol('tflive-store');

const emptyStatus = (): HubStatus => ({ logPath: null, logState: 'missing', live: false, lastLineWallMs: null, paused: null, diskMBPerHour: null, saveId: null, lastError: null });

export function wsUrl(loc: { protocol: string; host: string }): string {
  return `${loc.protocol === 'https:' ? 'wss' : 'ws'}://${loc.host}/ws`;
}

function isWire(x: unknown): x is WireMessage {
  const m = x as Partial<WireMessage> | null;
  return !!m && m.type === 'state' && typeof m.snapshot === 'object' && Array.isArray(m.alerts) && typeof m.status === 'object';
}

export function createStore(deps: StoreDeps): Store {
  const schedule = deps.setTimeout ?? ((fn: () => void, ms: number) => setTimeout(fn, ms));
  const state = reactive<AppState>({ snapshot: emptySnapshot(), alerts: [], status: emptyStatus(), conn: 'closed', mutes: [], received: false });
  let ws: WebSocketLike | null = null;
  let attempt = 0;
  let stopped = false;

  const open = () => {
    state.conn = 'connecting';
    const sock = deps.wsFactory(deps.url);
    ws = sock;
    sock.onopen = () => {
      attempt = 0;
      state.conn = 'open';
    };
    sock.onmessage = (ev) => {
      let msg: unknown;
      try {
        msg = JSON.parse(String(ev.data));
      } catch {
        return;
      }
      if (!isWire(msg)) return;
      state.snapshot = msg.snapshot;
      state.alerts = msg.alerts;
      state.status = msg.status;
      state.received = true;
    };
    sock.onerror = () => {};
    sock.onclose = () => {
      if (ws !== sock) return;
      state.conn = 'closed';
      ws = null;
      if (stopped) return;
      const delay = BACKOFF_MS[Math.min(attempt, BACKOFF_MS.length - 1)];
      attempt += 1;
      schedule(() => {
        if (!stopped) open();
      }, delay);
    };
  };

  const refreshMutes = async () => {
    const res = await deps.fetch('/api/mutes');
    if (res.ok) state.mutes = (await res.json()) as Mute[];
  };

  return {
    state,
    connect() {
      stopped = false;
      if (!ws) open();
    },
    close() {
      stopped = true;
      ws?.close();
    },
    refreshMutes,
    async mute(alert, kind) {
      await deps.fetch('/api/mutes', {
        method: 'POST',
        headers: { 'content-type': 'application/json' },
        body: JSON.stringify({ alertType: alert.type, entityId: alert.entityId, kind }),
      });
      await refreshMutes();
    },
    async unmute(alert) {
      const m = state.mutes.find((x) => x.alertType === alert.type && x.entityId === alert.entityId);
      if (!m) return;
      await deps.fetch(`/api/mutes/${m.id}`, { method: 'DELETE' });
      await refreshMutes();
    },
  };
}

export function useStore(): Store {
  const s = inject(STORE_KEY);
  if (!s) throw new Error('TFLive: brak magazynu stanu (provide STORE_KEY)');
  return s;
}
```

```ts
  });
```

```vue
import { computed, ref } from 'vue';
import { alertsFor, isRampUp, lineLoadNow } from '../lib/derive';
import { alertEmoji, cargoEmoji, modeEmoji, modeLabel, VEHICLE_STATE } from '../lib/emoji';

import { useStore } from '../state/store';

const { state } = useStore();
const sn = computed(() => state.snapshot);
type SortKey = 'result' | 'vehicles';
const sort = ref<{ key: SortKey; dir: 1 | -1 }>({ key: 'result', dir: 1 });
const selectedId = ref<number | null>(null);

const rows = computed(() =>
  sn.value.lines
    .map((l) => ({
      line: l,
      load: lineLoadNow(l, sn.value.vehicles),
      rampUp: sn.value.meta ? isRampUp(l, sn.value.meta.gameTime, sn.value.meta.dayMs) : false,
      alerts: alertsFor(state.alerts, 'line', l.id).filter((a) => !a.muted),
    }))
    .sort((a, b) => {
      const va = sort.value.key === 'result' ? a.line.operatingResult12m : a.line.vehicleIds.length;
      const vb = sort.value.key === 'result' ? b.line.operatingResult12m : b.line.vehicleIds.length;
      return (va - vb) * sort.value.dir;
    }),
);
const selected = computed<Line | null>(() => sn.value.lines.find((l) => l.id === selectedId.value) ?? rows.value[0]?.line ?? null);
const detail = computed(() => {
  const l = selected.value;
  if (!l) return null;
  const stops = l.stopIds.map((id) => sn.value.stops.find((s) => s.id === id)).filter((s) => !!s);
  const vehicles = sn.value.vehicles.filter((v) => v.lineId === l.id);
  const months = [...l.operatingResultByMonth].sort((a, b) => a.month - b.month);
  const max = Math.max(1, ...months.map((m) => Math.abs(m.value)));
  return {
    line: l,
    load: lineLoadNow(l, sn.value.vehicles),
    stops,
    vehicles,
    months: months.map((m) => ({ ...m, label: fmtMonthIndex(m.month), h: `${Math.max(6, (Math.abs(m.value) / max) * 100)}%`, pos: m.value >= 0 })),
    alerts: alertsFor(state.alerts, 'line', l.id),
  };
});

function toggle(key: SortKey) {
  sort.value = sort.value.key === key ? { key, dir: sort.value.dir === 1 ? -1 : 1 } : { key, dir: key === 'result' ? 1 : -1 };
```
